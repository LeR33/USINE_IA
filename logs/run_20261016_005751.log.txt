**app/database.py**
```python
import os
import aiosqlite

DEFAULT_DB_PATH = "todo.db"
DB_PATH = DEFAULT_DB_PATH

def get_db_path() -> str:
    """Récupère le chemin de la base de données depuis l'environnement ou utilise le défaut."""
    return os.getenv("DATABASE_URL", DEFAULT_DB_PATH)

async def init_db() -> None:
    """Initialise la base de données et crée la table tasks si elle n'existe pas."""
    db_file = get_db_path()
    async with aiosqlite.connect(db_file) as db:
        await db.execute(
            """
            CREATE TABLE IF NOT EXISTS tasks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                title TEXT NOT NULL,
                completed BOOLEAN NOT NULL DEFAULT 0
            )
            """
        )
        await db.commit()
```

```python
import aiosqlite
from app.database import get_db_path

async def create_task(title: str):
    """Insère une nouvelle tâche dans la base de données."""
    async with aiosqlite.connect(get_db_path()) as db:
        await db.execute("INSERT INTO tasks (title, completed) VALUES (?, 0)", (title,))
        await db.commit()

async def get_tasks():
    """Récupère toutes les tâches de la base de données."""
    async with aiosqlite.connect(get_db_path()) as db:
        db.row_factory = aiosqlite.Row
        async with db.execute("SELECT * FROM tasks") as cursor:
            rows = await cursor.fetchall()
            return [dict(row) for row in rows]

async def toggle_task(task_id: int):
    """Inverse le statut 'completed' d'une tâche."""
    async with aiosqlite.connect(get_db_path()) as db:
        await db.execute("UPDATE tasks SET completed = NOT completed WHERE id = ?", (task_id,))
        await db.commit()
```

**app/main.py**
```python
from contextlib import asynccontextmanager
from pathlib import Path
from fastapi import FastAPI, Form, Request
from fastapi.responses import HTMLResponse, RedirectResponse
from fastapi.templating import Jinja2Templates

from app.database import init_db
from app.crud import create_task, get_tasks, toggle_task

@asynccontextmanager
async def lifespan(app: FastAPI):
    """Gère le cycle de vie de l'application (initialisation DB)."""
    await init_db()
    yield

app = FastAPI(title="Gestionnaire de Tâches", lifespan=lifespan)

# Configuration des templates Jinja2
templates_dir = Path(__file__).parent / "templates"
templates = Jinja2Templates(directory=str(templates_dir))

@app.get("/", response_class=HTMLResponse)
async def read_root(request: Request):
    """Affiche la liste des tâches."""
    tasks = await get_tasks()
    return templates.TemplateResponse(
        request=request,
        name="index.html",
        context={"tasks": tasks},
    )

@app.post("/tasks")
async def add_task(title: str = Form(...)):
    """Ajoute une tâche via le formulaire."""
    clean_title = title.strip()
    if clean_title:
        await create_task(clean_title)
    return RedirectResponse(url="/", status_code=303)

@app.post("/tasks/{task_id}/toggle")
async def toggle_task_status(task_id: int):
    """Change le statut d'une tâche via un bouton."""
    await toggle_task(task_id)
    return RedirectResponse(url="/", status_code=303)
```

import os
import pytest
from httpx import ASGITransport, AsyncClient
from app.main import app
from app.database import init_db, get_db_path

@pytest.mark.asyncio
async def test_full_workflow(tmp_path, monkeypatch):

    test_db = tmp_path / "test_todo.db"
    monkeypatch.setenv("DATABASE_URL", str(test_db))
    await init_db()

    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://testserver") as client:
        # 1. Vérifier que la page d'accueil charge
        response = await client.get("/")
        assert response.status_code == 200
        assert "Ma Liste de Tâches" in response.text

        # 2. Ajouter une tâche

        assert response.status_code == 303
        
        # 3. Vérifier la présence de la tâche

        assert "Tester l'application" in response.text

        task_id = tasks[0]["id"]
